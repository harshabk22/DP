class MessMateNativeBridge(private val activity: MainActivity) {
    @JavascriptInterface
    fun scheduleDelayedNotification(mealId: String, prompt: String, optionsJson: String, delaySeconds: Long) {
        Handler(Looper.getMainLooper()).postDelayed({
            activity.showClarificationNotification(mealId, prompt)
        }, delaySeconds * 1000L)
    }

    @JavascriptInterface
    fun setServerUrl(url: String) { activity.saveAndReloadUrl(url) }

    @JavascriptInterface
    fun isNativeAndroid(): Boolean = true
}

View Cosine Matching Function (Lines 112-130) 
def compute_tableware_similarity(vec1: np.ndarray, vec2: np.ndarray) -> float:
    dot = np.dot(vec1, vec2)
    norm = np.linalg.norm(vec1) * np.linalg.norm(vec2)
    return float(dot / norm) if norm > 0 else 0.0
